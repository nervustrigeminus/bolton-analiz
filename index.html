    // Norm Tablosunu Çiz

    function renderNormTable(highlightVal = 0) {

      const tbody = document.getElementById('normTableBody');

      tbody.innerHTML = '';

      const list = currentTableMode === 12 ? normData12 : normData6;


      let closestDiff = 999;

      let closestUpper = -1;


      if (highlightVal > 0) {

        list.forEach(row => {

          const diff = Math.abs(row.upper - highlightVal);

          if (diff < closestDiff) {

            closestDiff = diff;

            closestUpper = row.upper;

          }

        });

      }


      list.forEach(row => {

        const isClosest = highlightVal > 0 && Math.abs(row.upper - closestUpper) < 0.1;

        const tr = document.createElement('tr');

        if (isClosest) {

          tr.className = "bg-teal-950/70 text-teal-200 font-bold border-l-4 border-l-teal-400";

        } else {

          tr.className = "hover:bg-slate-800/40 transition";

        }


        tr.innerHTML = `

          <td class="p-2 border-r border-slate-800/60">${row.upper.toFixed(1)} ${isClosest ? '<span class="text-[9px] bg-teal-500 text-slate-950 px-1 py-0.2 rounded font-sans ml-1">Hastaya En Yakın</span>' : ''}</td>

          <td class="p-2">${row.lower.toFixed(1)}</td>

        `;

        tbody.appendChild(tr);

      });

    }
